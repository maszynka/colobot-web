# colobot-web WP4: WebGPU backend (emdawnwebgpu). Web build only.

# --- Embed WGSL sources into the binary --------------------------------------
# Every shaders/*.wgsl becomes Gfx::WGSL::<name> (const char*) in the generated
# header wgsl_sources.h. Editing a .wgsl re-runs CMake (CMAKE_CONFIGURE_DEPENDS).
file(GLOB COLOBOT_WGSL_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/*.wgsl")
list(SORT COLOBOT_WGSL_FILES)
set(_wgsl_header "// Generated by graphics/webgpu/CMakeLists.txt - do not edit\n#pragma once\n\nnamespace Gfx::WGSL\n{\n")
foreach(_wgsl IN LISTS COLOBOT_WGSL_FILES)
    get_filename_component(_name "${_wgsl}" NAME_WE)
    file(READ "${_wgsl}" _content)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_wgsl}")
    string(APPEND _wgsl_header "inline constexpr const char* ${_name} = R\"WGSL(${_content})WGSL\";\n")
endforeach()
string(APPEND _wgsl_header "} // namespace Gfx::WGSL\n")
set(_wgsl_out "${CMAKE_CURRENT_BINARY_DIR}/generated/wgsl_sources.h")
if(EXISTS "${_wgsl_out}")
    file(READ "${_wgsl_out}" _wgsl_old)
else()
    set(_wgsl_old "")
endif()
if(NOT _wgsl_old STREQUAL _wgsl_header)
    file(WRITE "${_wgsl_out}" "${_wgsl_header}")
endif()
target_include_directories(Colobot-Base PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

target_sources(Colobot-Base PRIVATE
    wgpu_device.cpp
    wgpu_device.h
    wgpu_framebuffer.cpp
    wgpu_framebuffer.h
    wgpu_pass_manager.cpp
    wgpu_pass_manager.h
    wgpu_pipeline_cache.cpp
    wgpu_pipeline_cache.h
    wgpu_ring_buffer.cpp
    wgpu_ring_buffer.h
    wgpu_scene.cpp
    wgpu_scene.h
    wgpu_texture.cpp
    wgpu_texture.h
    wgpu_util.cpp
    wgpu_util.h
    wgpu_ui_renderer.cpp
    wgpu_ui_renderer.h
    wgpu_terrain_renderer.cpp
    wgpu_terrain_renderer.h
    wgpu_object_renderer.cpp
    wgpu_object_renderer.h
    wgpu_particle_renderer.cpp
    wgpu_particle_renderer.h
    wgpu_shadow_renderer.cpp
    wgpu_shadow_renderer.h
)
